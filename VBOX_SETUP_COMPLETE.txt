🚀 QUICK START (Copy & Paste)
═══════════════════════════════════════════════════════════════════════════

# Step 1: Read the quickstart guide
cat goad-vbox/docs/01-QUICKSTART.md

# Step 2: Check prerequisites
bash goad-vbox/scripts/check-requirements.sh

# Step 3: Create VirtualBox network
bash goad-vbox/scripts/network-setup.sh --variant full

# Step 4: Create VMs
bash goad-vbox/scripts/create-vms.sh \
  --variant full \
  --iso-2016 ~/Downloads/WinServer2016.iso \
  --iso-2019 ~/Downloads/WinServer2019.iso

# Step 5: [MANUAL] Install Windows on each VM
#   - Start each VM: VBoxManage startvm dc01 --type gui
#   - Follow Windows installation wizard
#   - Set hostname and static IP

# Step 6: [MANUAL] Enable WinRM on each VM
#   - Copy templates/ConfigureRemotingForAnsible.ps1 to VM
#   - Run as Administrator in PowerShell

# Step 7: Configure Ansible inventory
cp goad-vbox/inventory/hosts-template.ini goad-vbox/inventory/hosts.ini
# Edit with your VM IPs and passwords

# Step 8: Provision the lab
bash goad-vbox/scripts/apply-provisioning.sh \
  --variant full \
  --inventory goad-vbox/inventory/hosts.ini

# Step 9: Verify lab is ready
bash goad-vbox/scripts/health-check.sh \
  --inventory goad-vbox/inventory/hosts.ini

⚡ QUICK COMMANDS REFERENCE
═══════════════════════════════════════════════════════════════════════════

# Check if everything is installed
bash goad-vbox/scripts/check-requirements.sh

# See VirtualBox VMs
VBoxManage list vms

# Start a VM
VBoxManage startvm dc01 --type headless

# Stop a VM
VBoxManage controlvm dc01 poweroff

# Test Ansible connectivity
ansible -i goad-vbox/inventory/hosts.ini windows -m win_ping

# Check AD domain
ansible -i goad-vbox/inventory/hosts.ini dc01 -m win_command -a "Get-ADDomain"

# View provisioning logs
tail -f goad-vbox/logs/provision.log

# Cleanup everything (WARNING: deletes all VMs)
bash goad-vbox/scripts/cleanup-vms.sh --variant full --force
